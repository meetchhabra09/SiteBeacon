━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Authentication: Bearer Token Required

GET    /analytics/beacon/:id
       ├─ Get beacon analytics
       └─ Returns: metrics, incidents, dailyStats

POST   /analytics/beacon/:id/send-email
       ├─ Send report email
       └─ Returns: success message

GET    /analytics/beacon/:id/history
       ├─ Get status history (paginated)
       └─ Returns: history array, pagination

GET    /analytics/summary
       ├─ Summary for all user's beacons
       └─ Returns: all uptime %, downtime counts

GET    /analytics/preferences
       ├─ Get user preferences
       └─ Returns: preferences object

PUT    /analytics/preferences
       ├─ Update preferences
       └─ Accepts: emailReports, reportFrequency, reportDay, reportTime
